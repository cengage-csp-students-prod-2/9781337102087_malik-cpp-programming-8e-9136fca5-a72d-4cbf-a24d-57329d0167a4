    ASSERT_EQ(clock1.timeRemaining(), 68130);
}TEST(timeRemaining, 2) {

    ASSERT_EQ(clock1.timeElapsed(), 18270);
}TEST(timeElapsed, 2) {

    clockType clock1;

    ASSERT_EQ(clock1.timeDifference(clock2), 0);

    clockType clock1;

    ASSERT_EQ(clock1.timeDifference(clock2), 25235);
}
